import {spawn,spawnSync} from 'node:child_process';
import {mkdtempSync,readFileSync,writeFileSync,mkdirSync} from 'node:fs';
import {tmpdir} from 'node:os';
import {join,dirname} from 'node:path';
import {createServer} from 'node:net';
import assert from 'node:assert/strict';
const binary=process.argv[2];
assert(binary,'binary path required');
const home=mkdtempSync(join(tmpdir(),'factory-r1-product-'));
mkdirSync(join(home,'empty-capacity'));
const env={...process.env,BOBS_FACTORY_SENTRY_DISABLED:'1',BOBS_FACTORY_MIGRATION_SOURCE_CAPACITY_DIRECTORY:join(home,'empty-capacity')};
const reserve=createServer();await new Promise(r=>reserve.listen(0,'127.0.0.1',r));
const port=reserve.address().port;await new Promise(r=>reserve.close(r));
const base=`http://127.0.0.1:${port}`;
const worker=spawn(binary,['--home',join(home,'state'),'--port',String(port),'--no-open'],{env,stdio:['ignore','pipe','pipe']});
let logs='';worker.stdout.on('data',b=>logs+=b);worker.stderr.on('data',b=>logs+=b);
const sleep=ms=>new Promise(r=>setTimeout(r,ms));
const children=()=>spawnSync('ps',['-axo','pid=,ppid=,command='],{encoding:'utf8'}).stdout.split('\n').flatMap(line=>{const m=line.trim().match(/^(\d+)\s+(\d+)\s+(.*)$/);return m&&Number(m[2])===worker.pid&&m[3].includes('internal-schema-validator')?[{pid:Number(m[1]),command:m[3]}]:[]});
const post=(schema,signal)=>fetch(`${base}/api/workflows/step-config/validate`,{method:'POST',headers:{'content-type':'application/json','x-factory-request':'1','origin':base},body:JSON.stringify({schema}),signal}).then(async r=>({status:r.status,...await r.json()}));
const report={binary,home};
try {
 let ready=false;for(let i=0;i<120;i++){try{const r=await fetch(`${base}/api/version`);if(r.ok){report.identity=await r.json();ready=true;break;}}catch{}if(worker.exitCode!==null)throw Error(`worker exited ${worker.exitCode}: ${logs}`);await sleep(250);}assert(ready,logs);
 const build=JSON.parse(readFileSync(join(dirname(binary),'build.json'),'utf8'));assert.equal(build.commit,'807b963cfd2879ae466add6e041a8907eecffd22');assert.equal(build.dirty,false);
 for(const key of ['version','commit','dirty','target','resourceDigest'])assert.equal(report.identity.runtime[key],build[key]);
 assert.equal(report.identity.runtime.packaged,true);
 const helper=spawnSync(binary,['internal-schema-validator'],{env,input:JSON.stringify({schema:{type:'integer',minimum:1},value:3,validate:true}),encoding:'utf8',timeout:5000});assert.equal(helper.status,0,helper.stderr);assert.deepEqual(JSON.parse(helper.stdout),{schemaIssues:[],issues:[]});report.privateDispatch=true;
 const useful=await post({type:'object',properties:{title:{$ref:'#/definitions/name'}},required:['title'],definitions:{name:{type:'string',minLength:1}}});assert.deepEqual(useful,{status:200,valid:true,issues:[]});report.usefulSchema=true;
 for(const [schema,message] of [[{$ref:'#'},'reference cycles'],[{type:'string',pattern:'^(a+)+$'},'unsafe pattern'],[{type:'object',patternProperties:{'^(a+)+$':{type:'string'}}},'unsafe pattern']]){const result=await post(schema);assert.equal(result.valid,false);assert(result.issues.some(i=>i.includes(message)));}report.unsafeRejected=true;
 const expensive={enum:Array.from({length:4990},(_,i)=>({i}))};
 const seen=new Map();let probes=0,maxPing=0,stopped=false;const started=Date.now();let resolved=false;
 const request=post(expensive).finally(()=>resolved=true);
 while(!resolved){for(const child of children()){seen.set(child.pid,child);if(!stopped){process.kill(child.pid,'SIGSTOP');stopped=true;}}const at=Date.now();const ping=await fetch(`${base}/api/workflows/step-config/catalog`);assert.equal(ping.status,200);maxPing=Math.max(maxPing,Date.now()-at);probes++;await sleep(40);}
 const result=await request;assert.equal(result.status,200);assert.equal(result.valid,false);assert(result.issues.some(i=>/execution limit|wall limit|resource|watchdog/i.test(i)),JSON.stringify(result));assert(probes>=2);assert(maxPing<1000,`HTTP latency ${maxPing}`);assert(Date.now()-started<4000);assert(seen.size>0,'Did not observe real helper child');assert(stopped,'Did not inject stuck child');assert.deepEqual(children(),[],'Child still alive after result');report.expensiveCompile={result,elapsedMs:Date.now()-started,probes,maxPingMs:maxPing,faultInjection:'SIGSTOP own validator child to simulate a nonresponsive engine; production parent deadline must SIGKILL',observedChildren:[...seen.values()],childrenAfterResponse:children()};
 const controller=new AbortController();const cancel=post(expensive,controller.signal);for(let i=0;i<50&&!children().length;i++)await sleep(10);report.cancelChildren=children();assert(report.cancelChildren.length>0,'No cancellation helper');for(const child of report.cancelChildren)process.kill(child.pid,'SIGSTOP');controller.abort();await assert.rejects(cancel,error=>error.name==='AbortError');for(let i=0;i<40&&children().length;i++)await sleep(25);assert.deepEqual(children(),[],'Cancelled child remains alive');const after=await post({type:'boolean'});assert.equal(after.valid,true);report.abortReclaimed=true;
 console.log(JSON.stringify(report,null,2));
} finally {
 writeFileSync(join(home,'worker.log'),logs);
 worker.kill('SIGTERM');await new Promise(resolve=>{if(worker.exitCode!==null)return resolve();worker.once('exit',resolve);setTimeout(()=>worker.kill('SIGKILL'),5000).unref();});
}
