import {spawn,execFile} from 'node:child_process';
import {writeFileSync} from 'node:fs';
import assert from 'node:assert/strict';
const jobs=[];
async function measure(name){return await new Promise((resolve,reject)=>{
const begin=performance.now(), child=spawn('/tmp/ticket136-r11-startup-prototype/'+name,['internal-schema-validator'],{stdio:['pipe','pipe','pipe']});
let stdout='',stderr='',rss=[],firstByteMs;
child.stdout.on('data',b=>{firstByteMs??=performance.now()-begin;stdout+=b}); child.stderr.on('data',b=>stderr+=b);
let busy=false;const probe=setInterval(()=>{if(busy)return;busy=true;execFile('ps',['-o','rss=','-p',String(child.pid)],{encoding:'utf8'},(e,out)=>{if(out.trim())rss.push(Number(out.trim()));busy=false})},10);
child.stdin.end(JSON.stringify({schema:{type:'boolean'},validate:false}));
child.once('error',reject);child.once('close',(code,signal)=>{clearInterval(probe);assert.equal(code,0,stderr);assert.equal(signal,null);assert.deepEqual(JSON.parse(stdout),{schemaIssues:[],issues:[]});resolve({name,elapsedMs:performance.now()-begin,firstByteMs,maxRssKiB:Math.max(...rss),samples:rss.length})});
})}
for(let i=0;i<12;i++)for(const name of i%2?['lazy','eager']:['eager','lazy'])jobs.push(await measure(name));
const summary=Object.fromEntries(['eager','lazy'].map(name=>{const a=jobs.filter(x=>x.name===name);return [name,{runs:a.length,elapsedMs:{min:Math.min(...a.map(x=>x.elapsedMs)),max:Math.max(...a.map(x=>x.elapsedMs)),mean:a.reduce((s,x)=>s+x.elapsedMs,0)/a.length},maxRssKiB:Math.max(...a.map(x=>x.maxRssKiB))}]}));
writeFileSync('/tmp/ticket136-r11-startup-comparison-results.json.txt',JSON.stringify({scope:'diagnostic wrappers compiled from current app graph with identical empty resource defines; NOT final binary acceptance',jobs,summary},null,2));console.log(JSON.stringify(summary,null,2));
