#!/usr/bin/env python3
"""Stage reviewed evidence; --apply is the sole checkout-writing mode.

Default produces draft Markdown outside checkout. Final mode fails closed unless
all eight complete gates, source/artifacts and final evidence identities agree.
"""
import argparse, hashlib, json, pathlib, shlex, shutil, subprocess

ROOT = pathlib.Path('/Users/jappy/.t3/worktrees/bobs-factory/ticket-136-agent-step-recovery')
SOURCE = '7106f734ca6fc4066448e2766495f5bacb4fa0d9'
HERE = pathlib.Path(__file__).resolve().parent
VALIDATION = pathlib.Path('/tmp/ticket136-r2-validation-7106f734')
F1 = pathlib.Path('/tmp/ticket136-r2-f1-7106f734')
BINARY = pathlib.Path('/tmp/ticket136-binary-r2-7106f734/evidence')
REPORT = pathlib.Path('apps/f1/test-drives/2026-10-21-configurable-agent-step-r2.md')
ASSETS = pathlib.Path('apps/f1/test-drives/assets/2026-10-21-configurable-agent-step-r2')
GATES = {'typecheck', 'lint', 'packages', 'apps-vitest', 'desktop', 'scripts', 'audit', 'workflow-lint'}


def require(condition, message):
    if not condition:
        raise RuntimeError(message)


def read_json(path):
    return json.loads(path.read_text())


def digest(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def outside_checkout(path):
    require(not path.resolve().is_relative_to(ROOT.resolve()), 'Staging output must be outside checkout')


def copy_text(source, destination):
    require(source.is_file() and not source.is_symlink(), f'Not a regular evidence file: {source}')
    data = source.read_bytes()
    data.decode('utf-8')  # Reject compiled executables/archives; preserve text bytes.
    require(not source.name.endswith(('.tar.gz', '.wasm', '.zip')), f'Binary/archive refused: {source}')
    destination.parent.mkdir(parents=True, exist_ok=True)
    destination.write_bytes(data)


def verify_source(manifest):
    require(subprocess.check_output(['git', 'rev-parse', 'HEAD'], cwd=ROOT, text=True).strip() == SOURCE, 'Unexpected source HEAD')
    require(not subprocess.check_output(['git', 'status', '--porcelain'], cwd=ROOT, text=True).strip(), 'Checkout must remain clean before assembly')
    require(manifest['source'] == SOURCE, 'Manifest source mismatch')
    for name, expected in manifest['productionInputs'].items():
        require(digest(ROOT / name) == expected, f'Source input drift: {name}')
    actual_roots = {p.relative_to(ROOT).as_posix() for prefix in ['apps', 'packages'] for p in (ROOT/prefix).glob('*/dist') if p.exists()}
    require(actual_roots == set(manifest['artifacts']), 'Generated artifact root set changed')
    for directory, files in manifest['artifacts'].items():
        artifact_root = ROOT / directory
        require(artifact_root.is_dir() and not artifact_root.is_symlink(), f'Invalid artifact root: {directory}')
        entries = list(artifact_root.rglob('*'))
        require(not any(path.is_symlink() for path in entries), f'Artifact symlink in {directory}')
        actual_files = {path.relative_to(artifact_root).as_posix() for path in entries if path.is_file()}
        actual_directories = {path.relative_to(artifact_root).as_posix() for path in entries if path.is_dir()}
        expected_directories = {parent.as_posix() for name in files for parent in pathlib.PurePosixPath(name).parents if parent.as_posix() != '.'}
        require(actual_files == set(files), f'Artifact file set changed: {directory}')
        require(actual_directories == expected_directories, f'Artifact directory set changed: {directory}')
        for name, expected in files.items():
            require(digest(ROOT / directory / name) == expected, f'Artifact drift: {directory}/{name}')


def verify_gates():
    gates = read_json(VALIDATION / 'final-gates.json.txt')
    require(gates['identity']['source'] == SOURCE, 'Gate source mismatch')
    checks = gates['checks']
    require(len(checks) == 8 and {x['name'] for x in checks} == GATES, 'All eight complete gates must be present')
    for check in checks:
        require(check['exitCode'] == 0, f'Nonzero gate: {check["name"]}')
        log = VALIDATION / f'final-{check["name"]}.log'
        require(read_json_line(log.read_text().splitlines()[-1])['exitCode'] == 0, f'Missing terminal success: {log}')
    return checks


def read_json_line(line):
    return json.loads(line)


def verify_packets(ui, ui_receipt):
    summary = read_json(F1 / 'summary.json.txt')
    require(summary['revision'] == SOURCE and summary['exitCode'] == 0 and summary['result'] == 'PASS', 'Final F1 did not pass exact source')
    require(summary['paidCalls'] == 0 and summary['buildSha256'] == summary['postDriveBuildSha256'], 'F1 identity/cost mismatch')
    require(len(summary['results']) == 2, 'Both Factory runs required')
    for result in summary['results']:
        require(result['status'] == 'completed' and result['approval'] == 'approve', 'Factory approval/completion mismatch')
        require(len(result['sequence']) == 21 and result['activityCount'] == 23 and result['commentCount'] == 24, 'Unexpected final Factory counts')
    for name in ['before-identity.json.txt', 'after-identity.json.txt']:
        identity = read_json(F1 / name)
        require(identity['source'] == SOURCE and identity['pass'] and not identity['status'], 'F1 source/artifact proof mismatch')
    binary = read_json(BINARY / 'results.json.txt')
    require(binary['source'] == SOURCE and not binary['dirty'], 'Binary source mismatch')
    require(binary['tight100']['ordinaryRequests'] == 100 and binary['tight100']['ordinaryInvalidResponses'] == 0, 'Tight ordinary loop failed')
    strict = binary['strict']
    require(strict['compileRequests'] == strict['evaluations'] == 100 and strict['cancellations'] == strict['postCancelCompiles'] == strict['postCancelEvaluations'] == 10 and strict['result'] == 'PASS', 'Strict binary stress incomplete')
    require(binary['canonicalSmoke'] == binary['isolatedInstallSafeguards'] == 'PASS', 'Binary smoke/install incomplete')
    proof = read_json(BINARY / 'postsmoke-input-proof.json.txt')
    require(proof['source'] == SOURCE and not proof['mismatches'] and not proof['gitStatus'], 'Binary post-input proof mismatch')
    dependency = read_json(VALIDATION / 'dependency-identity-final.json.txt')
    require(dependency['source'] == SOURCE and dependency['lockedPackages'] == 892 and dependency['installedIdentities'] == 950, 'Dependency identity count mismatch')
    require(dependency['sameIdentitiesAsAcceptedCleanup'], 'Dependency cleanup identity mismatch')
    require(dependency['installedGraphSha256'] == digest(VALIDATION / 'installed-graph.json.txt'), 'Current graph bytes mismatch')
    require('Recheck final graph after gates' not in dependency.get('graphSnapshotSource', ''), 'Await final dependency snapshot refresh')
    receipt = read_json(ui_receipt)
    require(receipt['source'] == SOURCE and receipt['visuallyInspected'] and receipt['screenshotCount'] >= 26, 'Final UI review receipt incomplete')
    require(pathlib.Path(receipt['directory']).resolve() == ui.resolve(), 'UI receipt belongs to another directory')
    require(len([name for name in receipt['files'] if name.endswith('.png')]) == receipt['screenshotCount'], 'UI screenshot count mismatch')
    ui_manifest = read_json(ui / 'manifest.json.txt')
    require(ui_manifest['source'] == SOURCE and ui_manifest['screenshotCount'] == receipt['screenshotCount'] and len(ui_manifest['screenshots']) == receipt['screenshotCount'], 'UI manifest/receipt count mismatch')
    for name, expected in receipt['files'].items():
        rel = pathlib.Path(name)
        require(not rel.is_absolute() and '..' not in rel.parts, 'Unsafe UI path')
        require(digest(ui / rel) == expected, f'UI file changed: {name}')
    return receipt


def copy_tree_text(source, destination):
    for path in sorted(source.rglob('*')):
        if path.is_dir():
            continue
        copy_text(path, destination / path.relative_to(source))


def make_gate_table(checks):
    rows = ['| Command | Result | Complete log |', '| --- | --- | --- |']
    rows.append('| `pnpm build` (clean generated output) | PASS | [Build](final-build.log) |')
    for check in checks:
        rows.append(f'| `{shlex.join(check["command"])}` | PASS (exit0) | [{check["name"]}](final-{check["name"]}.log) |')
    return '\n'.join(rows)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--out', type=pathlib.Path, required=True)
    parser.add_argument('--final', action='store_true')
    parser.add_argument('--apply', action='store_true')
    parser.add_argument('--ui', type=pathlib.Path)
    parser.add_argument('--ui-receipt', type=pathlib.Path)
    args = parser.parse_args()
    outside_checkout(args.out)
    require(not args.out.exists(), 'Choose a fresh staging directory')
    require(not args.apply or args.final, '--apply requires final verification')
    checks = []
    if args.final:
        require(args.ui and args.ui_receipt, 'Final UI directory and normalized receipt required')
        manifest = read_json(VALIDATION / 'final-build-manifest.json.txt')
        verify_source(manifest)
        checks = verify_gates()
        count_record = read_json(VALIDATION / 'test-counts.json.txt')
        require(count_record['source'] == SOURCE and count_record['gatesSha256'] == digest(VALIDATION / 'final-gates.json.txt'), 'Complete test-count record is stale/missing')
        for name, expected in count_record['logs'].items():
            require(digest(VALIDATION / name) == expected, f'Test-count source log changed: {name}')
        receipt = verify_packets(args.ui, args.ui_receipt)
        status = 'Complete clean build and all eight complete gates PASS; exact commands and logs below.'
        ui_section = (HERE / 'ui-section.template.md').read_text().replace('{{UI_SCREENSHOT_COUNT}}', str(receipt['screenshotCount']))
        require('{{' not in ui_section, 'UI section is still a draft')
    else:
        status = 'DRAFT — complete gate packet is pending; no complete-gate success is claimed.'
        ui_section = (HERE / 'ui-section.template.md').read_text().replace('{{UI_SCREENSHOT_COUNT}}', str(read_json(HERE / 'ui-assembly-identity.json.txt')['screenshotCount']) if (HERE / 'ui-assembly-identity.json.txt').exists() else 'pending') if (HERE / 'ui-section.template.md').exists() else 'DRAFT — final UI packet pending.'
    report = (HERE / 'report.template.md').read_text().replace('{{GATE_STATUS}}', status).replace('{{UI_SECTION}}', ui_section)
    index = (HERE / 'validation.template.md').read_text().replace('{{GATE_STATUS}}', status).replace('{{GATE_TABLE}}', make_gate_table(checks) if checks else 'PENDING — all eight final exits must be recorded as 0 before assembly.')
    index = index.replace('{{TEST_COUNTS}}', (VALIDATION / 'test-counts.md').read_text() if args.final else 'PENDING — complete per-package/app/desktop/script terminal counts are generated only after all eight exits0.')
    report_out = args.out / REPORT
    validation_out = args.out / ASSETS / 'validation'
    report_out.parent.mkdir(parents=True, exist_ok=True)
    validation_out.mkdir(parents=True, exist_ok=True)
    report_out.write_text(report)
    (validation_out / 'README.md').write_text(index)
    if args.final:
        for path in sorted(VALIDATION.iterdir()):
            if path.is_file() and (path.name.endswith('.log') or path.name.endswith('.json.txt')):
                copy_text(path, validation_out / path.name)
        copy_text(VALIDATION / 'test-counts.md', validation_out / 'test-counts.md')
        copy_text(HERE / 'parse-test-counts.py', validation_out / 'parse-test-counts.py.txt')
        copy_text(HERE / 'prompt-fixture-identity.json.txt', validation_out / 'prompt-fixture-identity.json.txt')
        copy_text(pathlib.Path('/tmp/ticket136-root-evidence-proof.py'), validation_out / 'root-evidence-proof.py.txt')
        copy_tree_text(BINARY, validation_out / 'binary-final')
        f1_names = ['F1-REPORT.md', 'factory-f1.log', 'summary.json.txt', 'results.json.txt', 'stock-activities.json.txt', 'renamed-activities.json.txt', 'stock-comments.json.txt', 'renamed-comments.json.txt', 'before-identity.json.txt', 'after-identity.json.txt', 'build-manifest.json.txt', 'rule-results.json.txt', 'node24.log', 'compiled-bun.log', 'rule-node-build.log', 'rule-bun-build.log']
        for name in f1_names:
            copy_text(F1 / name, args.out / ASSETS / name)
        copy_text(ROOT / 'packages/edge-worker/test/fixtures/rule-budget-smoke.ts', args.out / ASSETS / 'rule-budget-smoke.ts.txt')
        copy_text(args.ui_receipt, args.out / ASSETS / 'ui-assembly-identity.json.txt')
        normalization = {'originalPacket': str(args.ui), 'reason': 'Raw fixture source preserved as .mjs.txt so evidence is not a new linted runtime script.', 'renamedFiles': {'legacy-ui-fixture.mjs': 'legacy-ui-fixture.mjs.txt', 'modern-ui-fixture.mjs': 'modern-ui-fixture.mjs.txt'}, 'documentationChange': 'Copied UI README links receive these two filename suffixes; original source packet hashes remain in ui-assembly-identity.json.txt.'}
        (args.out / ASSETS / 'ui-name-mapping.json.txt').write_text(json.dumps(normalization, indent=2) + '\n')
        for path in sorted(args.ui.rglob('*')):
            if path.is_dir():
                continue
            rel = path.relative_to(args.ui)
            if path.name.endswith('.png'):
                require(path.read_bytes().startswith(b'\x89PNG\r\n\x1a\n'), f'Invalid PNG: {path}')
                target = args.out / ASSETS / 'ui' / rel
                target.parent.mkdir(parents=True, exist_ok=True)
                shutil.copyfile(path, target)
            else:
                target = args.out / ASSETS / 'ui' / rel
                if path.suffix == '.mjs':
                    target = target.with_name(target.name + '.txt')
                copy_text(path, target)
                if path.name == 'README.md':
                    text = target.read_text().replace('(legacy-ui-fixture.mjs)', '(legacy-ui-fixture.mjs.txt)').replace('(modern-ui-fixture.mjs)', '(modern-ui-fixture.mjs.txt)')
                    target.write_text(text)
        for source, label in [(pathlib.Path('/tmp/ticket136-r2-final-validation'), 'gates'), (pathlib.Path('/tmp/ticket136-r2-final-f1'), 'F1')]:
            for path in sorted(source.rglob('*')):
                if not path.is_file() or path.is_symlink():
                    continue
                if not (path.name.endswith(('.log', '.md', '.json.txt'))):
                    continue  # excludes compiled fixture binaries and generated JS bundles
                copy_text(path, validation_out / 'preliminary-96b' / label / path.relative_to(source))
        focused = ['ticket136-r10-before.json.txt', 'ticket136-r10-own-rule-repro.mjs', 'ticket136-r10-report.md', 'ticket136-r10-after-final.json.txt', 'ticket136-r2-generic-architecture-repro.mjs', 'ticket136-r2-generic-question-repro.mjs', 'ticket136-r2-legacy-presentation-before.mjs', 'ticket136-r2-legacy-presentation-after.mjs', 'ticket136-r2-render-legacy-before.mjs', 'ticket136-r2-generic-architecture-before.json', 'ticket136-r2-generic-question-before.json', 'ticket136-r2-legacy-presentation-before.json', 'ticket136-r2-legacy-presentation-after.json', 'ticket136-r2-legacy-navigation-red.log', 'ticket136-r2-legacy-navigation-green.log', 'ticket136-r2-legacy-progress-red.log', 'ticket136-r2-legacy-progress-green.log', 'ticket136-r2-legacy-final-page-red.json', 'ticket136-r2-legacy-final-page-green.json', 'ticket136-r2-legacy-final-page-red.log', 'ticket136-r2-legacy-final-page-green.log']
        for name in focused:
            source = pathlib.Path('/tmp') / name
            require(source.exists(), f'Focused reproduction missing: {name}')
            copy_text(source, validation_out / 'focused-before-after' / (name if name.endswith('.txt') else name + '.txt'))
        copy_text(HERE / 'source-head-bridge.md', validation_out / 'source-head-bridge.md')
        copy_text(HERE / 'verify-head-bridge.py', validation_out / 'verify-head-bridge.py.txt')
        copy_text(HERE / 'assemble-r2.py', validation_out / 'assemble-r2.py.txt')
        files = {str(p.relative_to(args.out)): digest(p) for p in sorted(args.out.rglob('*')) if p.is_file()}
        (args.out / 'staging-files.json.txt').write_text(json.dumps({'source': SOURCE, 'files': files}, indent=2) + '\n')
    if args.apply:
        require(not (ROOT / REPORT).exists() and not (ROOT / ASSETS).exists(), 'Destination exists; no overwrite is permitted')
        verify_source(read_json(VALIDATION / 'final-build-manifest.json.txt'))
        shutil.copytree(args.out / ASSETS, ROOT / ASSETS)
        shutil.copyfile(report_out, ROOT / REPORT)
        print('Applied only declared new report/assets paths; no commit/push/merge performed.')
    print(('FINAL' if args.final else 'DRAFT') + ' staged outside checkout: ' + str(args.out))


if __name__ == '__main__':
    main()
